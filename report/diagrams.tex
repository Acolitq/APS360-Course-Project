% ---------- Baselines ----------
\begin{figure}[h]
\centering
\begin{tikzpicture}[node distance=3.5mm]
  \node[io] (in) {Board\\$22{\times}8{\times}8$};
  % heuristic
  \node[blk, right=6mm of in, yshift=7mm, fill=gray!20, text width=34mm] (h)
    {\textbf{Capture heuristic} (no learning):\\capture highest-value piece,\\else random legal move};
  % linear
  \node[lin, right=6mm of in, yshift=-7mm] (fl) {Flatten\\1408};
  \node[lin, right=of fl] (fc) {Linear\\$1408{\to}128$};
  \node[lin, right=of fc, text width=19mm] (sum) {$s(f,t)=a_f+b_t$\\(64 from + 64 to)};
  \node[out, right=7mm of sum, yshift=7mm] (o) {Mask illegal\\$\to$ softmax\\over moves};
  \draw[arr] (in) |- (h); \draw[arr] (in) |- (fl);
  \draw[arr] (fl) -- (fc); \draw[arr] (fc) -- (sum);
  \draw[arr] (sum) -| (o); \draw[arr] (h) -| (o);
\end{tikzpicture}
\caption{Baselines. The linear model (\linParams{} parameters) scores a move as a from-square term plus a
to-square term, both linear in the input, and is trained with the same masked cross-entropy as the CNN.}
\label{fig:baselines}
\end{figure}
